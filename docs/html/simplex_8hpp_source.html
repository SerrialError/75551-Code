36 static std::optional<std::vector<double>> solve(const std::vector<std::vector<double>>& A_in,
37 const std::vector<double>& b_in,
38 const std::vector<double>& c_in) {
39 const double EPS = 1e-9;
40 int m = (int)A_in.size();
41 int n = (int)c_in.size();
42 if (m == 0 || n == 0) throw std::invalid_argument("Empty LP input");
43
44 // ---- 1) Handle unrestricted original variables by splitting:
45 // For each original x_j (free), create x_j_pos and x_j_neg, both >= 0, with x_j = pos - neg.
46 // New variable count:
47 int n2 = 2 * n;
48
49 // Build A' (m x n2) and c' (n2)
50 std::vector<std::vector<double>> A(m, std::vector<double>(n2, 0.0));
51 std::vector<double> c(n2, 0.0);
52 for (int i = 0; i < m; ++i) {
53 for (int j = 0; j < n; ++j) {
54 A[i][2*j] = A_in[i][j]; // x_j_pos coef
55 A[i][2*j + 1] = -A_in[i][j]; // x_j_neg coef (because x = pos - neg)
56 }
57 }
58 for (int j = 0; j < n; ++j) {
59 c[2*j] = c_in[j]; // objective c^T (pos)
60 c[2*j + 1] = -c_in[j]; // objective for neg part is -c
61 }
62
63 // ---- 2) Build constraint rows with slacks/surplus and possibly artificials.
64 // We'll convert each constraint (originally A_in x <= b_in) into an equality using:
65 // - if b >= 0: keep <= and add slack (makes easy basic var)
66 // - if b < 0: multiply row by -1 => it becomes >= with new rhs >= 0
67 // For >= constraints we add a surplus (-s) and an artificial variable (a) which will be in basis.
68 //
69 // We'll track columns: [ original_vars (n2) | slacks | surplus (not kept separately) | artificials ] + RHS
70
71 std::vector<int> rowType(m, 0); // 0: originally <= and b>=0 (slack), 1: >= (we will add artificial), 2: equality (artificial)
72 std::vector<std::vector<double>> rows(m);
73 std::vector<double> rhs(m);
74
75 for (int i = 0; i < m; ++i) {
76 // copy A row
77 rows[i] = A[i];
78 double b = b_in[i];
79 if (b >= 0) {
80 // keep row as <=
81 rowType[i] = 0; // slack
82 rhs[i] = b;
83 } else {
84 // multiply by -1 => becomes >= with positive rhs
85 rowType[i] = 1; // will need surplus + artificial
86 for (double &val : rows[i]) val = -val;
87 rhs[i] = -b;
88 }
89 }
90
91 // Count slack and artificial vars
92 int slack_count = 0;
93 int art_count = 0;
94 for (int i = 0; i < m; ++i) {
95 if (rowType[i] == 0) slack_count++;
96 else art_count++; // for >= we will add an artificial (and a surplus which is not basic)
97 }
98
99 int totalCols = n2 + slack_count + art_count; // exclude RHS
100 int slack_start = n2;
101 int art_start = n2 + slack_count;
102
103 // Build full tableau (m constraint rows + 1 objective row) with totalCols + 1 (RHS)
104 int cols = totalCols + 1;
105 std::vector<std::vector<double>> tab(m + 1, std::vector<double>(cols, 0.0));
106 // basis: for each row, which column is basic
107 std::vector<int> basis(m, -1);
108
109 // Fill constraint rows
110 int s_idx = 0;
111 int a_idx = 0;
112 for (int i = 0; i < m; ++i) {
113 // copy variable columns
114 for (int j = 0; j < n2; ++j) tab[i][j] = rows[i][j];
115
116 if (rowType[i] == 0) {
117 // add slack variable with coefficient +1 and make it basic
118 tab[i][slack_start + s_idx] = 1.0;
119 basis[i] = slack_start + s_idx;
120 s_idx++;
121 } else {
122 // rowType == 1 (>= after flipping): add surplus (coefficient -1) and an artificial (+1)
123 // we won't keep surplus as basic; artificial will be basic
124 // Surplus column is not tracked separately as a basic var; we simply put -1 in some column if we wanted.
125 // For simplicity, we omit creating an explicit surplus column and only add artificial column with +1
126 // and put a -1 where the surplus would be into the artificial's handling is standard.
127 // We'll add artificial var column and make it basic.
128 tab[i][art_start + a_idx] = 1.0; // artificial
129 basis[i] = art_start + a_idx;
130 a_idx++;
131 // Note: the surplus column could be added if you want explicit variable indexing of surplus vars.
132 }
133
134 // RHS
135 tab[i][cols - 1] = rhs[i];
136 }
137
138 // ---- Phase I setup: objective is minimize sum(artificials).
139 // We will do Phase I only if art_count > 0 (if art_count==0 then feasible start already).
140 auto pivot = [&](int prow, int pcol, int rowsTot, int colsTot, std::vector<std::vector<double>>& T, std::vector<int>& basisRef) {
141 double piv = T[prow][pcol];
142 if (std::abs(piv) < EPS) throw std::runtime_error("Numerical instability (pivot too small)");
143 // normalize pivot row
144 for (int j = 0; j < colsTot; ++j) T[prow][j] /= piv;
145 // eliminate other rows
146 for (int i = 0; i < rowsTot; ++i) {
147 if (i == prow) continue;
148 double fac = T[i][pcol];
149 if (std::abs(fac) < EPS) continue;
150 for (int j = 0; j < colsTot; ++j) T[i][j] -= fac * T[prow][j];
151 }
152 basisRef[prow] = pcol;
153 };
154
155 auto simplex_loop = [&](std::vector<std::vector<double>>& T, std::vector<int>& basisRef, int rowsTot, int colsTot) -> void {
156 // T has rowsTot rows, colsTot columns (including RHS at colsTot-1). last row is objective row.
157 while (true) {
158 // find entering column: most negative in objective row (we keep the same convention: T[last][j] < -EPS)
159 int last = rowsTot - 1;
160 int enter = -1;
161 for (int j = 0; j < colsTot - 1; ++j) {
162 if (T[last][j] < -EPS && (enter == -1 || T[last][j] < T[last][enter])) enter = j;
163 }
164 if (enter == -1) break; // optimal
165
166 // ratio test for leaving row
167 int leave = -1;
168 double bestRatio = 0.0;
169 for (int i = 0; i < rowsTot - 1; ++i) {
170 if (T[i][enter] > EPS) {
171 double ratio = T[i][colsTot - 1] / T[i][enter];
172 if (ratio >= -EPS && (leave == -1 || ratio < bestRatio - EPS || (std::abs(ratio - bestRatio) < EPS && basisRef[i] < basisRef[leave]))) {
173 leave = i;
174 bestRatio = ratio;
175 }
176 }
177 }
178 if (leave == -1) throw std::runtime_error("Unbounded LP (during simplex loop)");
179 pivot(leave, enter, rowsTot, colsTot, T, basisRef);
180 }
181 };
182
183 // If there are artificial vars, prepare Phase I objective
184 if (art_count > 0) {
185 // bottom row stores -c (as in your original code). For Phase I, objective we want to maximize - sum(artificials)
186 // so c_phase1_j = -1 for artificial columns, hence bottom entries = -c_phase1 = +1 for art columns.
187 int last = m;
188 for (int j = 0; j < cols; ++j) tab[last][j] = 0.0;
189
190 // set bottom row (initial -c_phase1)
191 for (int j = art_start; j < art_start + art_count; ++j) {
192 tab[last][j] = 1.0; // -c_phase1 = -(-1) = +1
193 }
194 tab[last][cols - 1] = 0.0;
195
196 // For every artificial that is currently BASIC (we set them as basic for those rows),
197 // we must zero its reduced cost by doing bottom += c_b * row, where c_b = -1 (coefficient in phase1 objective)
198 // c_b = -1 => bottom += -1 * row => bottom -= row
199 for (int i = 0; i < m; ++i) {
200 int bcol = basis[i];
201 if (bcol >= art_start && bcol < art_start + art_count) {
202 // c_b = -1
203 double c_b = -1.0;
204 for (int j = 0; j < cols; ++j) tab[last][j] += c_b * tab[i][j];
205 }
206 }
207
208 // Run Phase I simplex
209 simplex_loop(tab, basis, m + 1, cols);
210
211 double phase1_obj = tab[m][cols - 1]; // bottom RHS is the maximized value of (-sum(art))
212 // recall: Phase I objective w* = -phase1_obj
213 if (phase1_obj < -EPS) {
214 // phase1_obj negative means w* > 0 => infeasible
215 // throw std::runtime_error("LP is infeasible (Phase I found positive artificial sum)");
216 return std::nullopt;
217 }
218
219 // Remove artificial columns from tableau.
220 // If an artificial is in basis, try to pivot it out using a non-artificial nonzero column in that row.
221 for (int a = 0; a < art_count; ++a) {
222 int acol = art_start + a;
223 // If artificial column is basic in some row, remove it by pivoting if possible
224 int prow = -1;
225 for (int i = 0; i < m; ++i) if (basis[i] == acol) prow = i;
226 if (prow != -1) {
227 // find a nonzero column in this row that's not artificial to pivot on
228 int newcol = -1;
229 for (int j = 0; j < art_start; ++j) {
230 if (std::abs(tab[prow][j]) > EPS) {
231 newcol = j;
232 break;
233 }
234 }
235 if (newcol != -1) {
236 pivot(prow, newcol, m + 1, cols, tab, basis);
237 } else {
238 // entire row zero in non-art columns; artificial is zero variable and can be removed safely.
239 // mark basis as -1 and continue; the column will be removed below.
240 basis[prow] = -1;
241 }
242 }
243 }
244
245 // Now physically remove artificial columns by creating a new tableau without them
246 int newTotalCols = art_start; // drop everything from art_start .. art_start+art_count-1
247 int newCols = newTotalCols + 1;
248 std::vector<std::vector<double>> newTab(m + 1, std::vector<double>(newCols, 0.0));
249 std::vector<int> newBasis(m, -1);
250 for (int i = 0; i < m + 1; ++i) {
251 for (int j = 0; j < newTotalCols; ++j) {
252 newTab[i][j] = tab[i][j];
253 }
254 newTab[i][newCols - 1] = tab[i][cols - 1];
255 }
256 // translate basis indices (anything >= art_start now removed)
257 for (int i = 0; i < m; ++i) {
258 if (basis[i] == -1) newBasis[i] = -1;
259 else if (basis[i] < art_start) newBasis[i] = basis[i];
260 else newBasis[i] = -1; // should not happen after our pivot attempts; but leave -1 if so
261 }
262 tab.swap(newTab);
263 basis.swap(newBasis);
264 cols = newCols;
265 totalCols = newTotalCols;
266 // slack_start and art_start no longer needed
267 }
268
269 // ---- Phase II: set objective to original c (on transformed variables).
270 // bottom row stores -c (as your original code). Build bottom row and then fix it using basis.
271 int last_row = m;
272 for (int j = 0; j < cols; ++j) tab[last_row][j] = 0.0;
273 // c has length n2; we must copy to columns 0..n2-1 (some columns may have been removed if artificials existed)
274 for (int j = 0; j < (int)c.size() && j < cols - 1; ++j) tab[last_row][j] = -c[j];
275 tab[last_row][cols - 1] = 0.0;
276
277 // For each basic variable with index bcol and corresponding objective coefficient c_b (we need c vector extended by zeros for slacks),
278 // add c_b * row_i to bottom row to make reduced costs correct. c_b for slack columns is 0.
279 for (int i = 0; i < m; ++i) {
280 int bcol = basis[i];
281 double c_b = 0.0;
282 if (bcol >= 0 && bcol < (int)c.size()) c_b = c[bcol];
283 // else slack columns have c_b = 0
284 if (std::abs(c_b) > 0.0) {
285 for (int j = 0; j < cols; ++j) tab[last_row][j] += c_b * tab[i][j];
286 }
287 }
288
289 // Run Phase II simplex
290 simplex_loop(tab, basis, m + 1, cols);
291
292 // Extract solution for original variables (combine pos-neg pairs)
293 std::vector<double> x_orig(n, 0.0);
294 for (int j = 0; j < n; ++j) {
295 int col_pos = 2*j;
296 int col_neg = 2*j + 1;
297 double val_pos = 0.0, val_neg = 0.0;
298 int pivotRow = -1;
299 // find if col_pos is basic
300 for (int i = 0; i < m; ++i) {
301 if (basis[i] == col_pos) { pivotRow = i; break; }
302 }
303 if (pivotRow != -1) val_pos = tab[pivotRow][cols - 1];
304 // find if col_neg is basic
305 pivotRow = -1;
306 for (int i = 0; i < m; ++i) {
307 if (basis[i] == col_neg) { pivotRow = i; break; }
308 }
309 if (pivotRow != -1) val_neg = tab[pivotRow][cols - 1];
310 x_orig[j] = val_pos - val_neg;
311 }
312
313 return x_orig;
314 }
